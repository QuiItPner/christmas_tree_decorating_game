  // Simple Vanilla Interactive Behavior for Tree Studio
  (function initChristmasStudio() {
    // 1. Right Panel Tab Switcher
    const tabBtnItem = document.getElementById('tab-btn-item');
    const tabBtnTree = document.getElementById('tab-btn-tree');
    const panelItemProps = document.getElementById('panel-item-props');
    const panelTreeProps = document.getElementById('panel-tree-props');

    if (tabBtnItem && tabBtnTree && panelItemProps && panelTreeProps) {
      tabBtnItem.addEventListener('click', () => {
        tabBtnItem.className = 'inspector-tab flex-1 py-2 rounded-xl text-label-md font-label-md transition-all flex items-center justify-center gap-1.5 bg-primary-container text-on-primary shadow-sm';
        tabBtnTree.className = 'inspector-tab flex-1 py-2 rounded-xl text-label-md font-label-md transition-all flex items-center justify-center gap-1.5 bg-surface-cream text-on-surface-variant hover:text-pine-deep';
        panelItemProps.classList.remove('hidden');
        panelTreeProps.classList.add('hidden');
      });

      tabBtnTree.addEventListener('click', () => {
        tabBtnTree.className = 'inspector-tab flex-1 py-2 rounded-xl text-label-md font-label-md transition-all flex items-center justify-center gap-1.5 bg-primary-container text-on-primary shadow-sm';
        tabBtnItem.className = 'inspector-tab flex-1 py-2 rounded-xl text-label-md font-label-md transition-all flex items-center justify-center gap-1.5 bg-surface-cream text-on-surface-variant hover:text-pine-deep';
        panelTreeProps.classList.remove('hidden');
        panelItemProps.classList.add('hidden');
      });
    }

    // 2. Scale & Rotate live inspector updates
    const scaleSlider = document.getElementById('slider-scale');
    const scaleVal = document.getElementById('scale-value');
    const rotateSlider = document.getElementById('slider-rotate');
    const rotateVal = document.getElementById('rotate-value');
    const activeItem = document.getElementById('active-selected-item');

    function updateTransform() {
      if (!activeItem) return;
      const s = scaleSlider ? scaleSlider.value / 100 : 1;
      const r = rotateSlider ? rotateSlider.value : 0;
      activeItem.style.transform = `translate(0px, 0px) rotate(${r}deg) scale(${s})`;
      if (scaleVal) scaleVal.textContent = Math.round(s * 100) + '%';
      if (rotateVal) rotateVal.textContent = r + '°';
    }

    if (scaleSlider) scaleSlider.addEventListener('input', updateTransform);
    if (rotateSlider) rotateSlider.addEventListener('input', updateTransform);

    // 3. Color swatch clicks for active ornament
    const swatches = document.querySelectorAll('.color-swatch-bead');
    const ornamentFill = document.getElementById('selected-ornament-fill');
    swatches.forEach(bead => {
      bead.addEventListener('click', () => {
        swatches.forEach(b => b.classList.remove('ring-2', 'ring-primary', 'ring-offset-2'));
        bead.classList.add('ring-2', 'ring-primary', 'ring-offset-2');
        const color = bead.getAttribute('data-color');
        if (ornamentFill && color) {
          ornamentFill.setAttribute('fill', color);
        }
      });
    });

    // 4. Atmospheric snow toggle
    const snowToggle = document.getElementById('toggle-snow');
    const snowLayer = document.getElementById('snow-layer');
    if (snowToggle && snowLayer) {
      snowToggle.addEventListener('change', () => {
        snowLayer.style.opacity = snowToggle.checked ? '1' : '0';
      });
    }

    // 5. Atmospheric lights toggle
    const lightsToggle = document.getElementById('toggle-lights');
    const treeLights = document.getElementById('tree-twinkle-lights');
    if (lightsToggle && treeLights) {
      lightsToggle.addEventListener('change', () => {
        treeLights.style.opacity = lightsToggle.checked ? '1' : '0';
      });
    }

    // 6. Live greeting input binding
    const greetingInput = document.getElementById('input-greeting');
    const bannerText = document.getElementById('banner-text');
    if (greetingInput && bannerText) {
      greetingInput.addEventListener('input', (e) => {
        bannerText.textContent = e.target.value || 'Merry Christmas!';
      });
    }

    // 7. Category pills filter active state
    const catPills = document.querySelectorAll('.cat-pill');
    catPills.forEach(pill => {
      pill.addEventListener('click', () => {
        catPills.forEach(p => {
          p.className = 'cat-pill px-3 py-1 rounded-full text-label-sm font-label-sm transition-all bg-surface-cream text-on-surface-variant hover:text-pine-deep hover:bg-surface-container';
        });
        pill.className = 'cat-pill active px-3 py-1 rounded-full text-label-sm font-label-sm transition-all bg-primary-container text-on-primary shadow-sm';
      });
    });

    // 8. Toast notification helper
    const toast = document.getElementById('toast-pill');
    const toastText = document.getElementById('toast-text');
    function showToast(msg) {
      if (!toast) return;
      if (toastText) toastText.textContent = msg;
      toast.classList.remove('opacity-0', 'pointer-events-none');
      toast.classList.add('opacity-100');
      setTimeout(() => {
        toast.classList.add('opacity-0', 'pointer-events-none');
        toast.classList.remove('opacity-100');
      }, 2600);
    }

    const btnSave = document.getElementById('btn-save-design');
    if (btnSave) {
      btnSave.addEventListener('click', () => showToast('Design saved to your browser gallery!'));
    }

    const btnExport = document.getElementById('btn-export-tree');
    if (btnExport) {
      btnExport.addEventListener('click', () => showToast('Generating high-res festive PNG...'));
    }

    const btnRandom = document.getElementById('btn-randomize');
    if (btnRandom) {
      btnRandom.addEventListener('click', () => showToast('Magic wand: Decked the halls with new cheer!'));
    }

    // 9. Zoom level buttons
    let currentZoom = 100;
    const zoomText = document.getElementById('zoom-level');
    const canvasStage = document.getElementById('tree-container');
    const btnZoomIn = document.getElementById('btn-zoom-in');
    const btnZoomOut = document.getElementById('btn-zoom-out');

    if (btnZoomIn && btnZoomOut && zoomText && canvasStage) {
      btnZoomIn.addEventListener('click', () => {
        if (currentZoom < 140) currentZoom += 10;
        zoomText.textContent = currentZoom + '%';
        canvasStage.style.transform = `scale(${currentZoom / 100})`;
      });
      btnZoomOut.addEventListener('click', () => {
        if (currentZoom > 70) currentZoom -= 10;
        zoomText.textContent = currentZoom + '%';
        canvasStage.style.transform = `scale(${currentZoom / 100})`;
      });
    }

    // 10. Preview mode toggle (hide controls temporarily)
    const btnPreview = document.getElementById('btn-preview-mode');
    let previewActive = false;
    if (btnPreview) {
      btnPreview.addEventListener('click', () => {
        previewActive = !previewActive;
        const sidebars = document.querySelectorAll('aside');
        sidebars.forEach(sb => {
          sb.style.display = previewActive ? 'none' : '';
        });
        btnPreview.innerHTML = previewActive
          ? '<span class="material-symbols-outlined text-[16px]">edit</span><span>Exit Preview</span>'
          : '<span class="material-symbols-outlined text-[16px]">visibility</span><span class="hidden sm:inline">Preview</span>';
        showToast(previewActive ? 'Preview Mode: Controls hidden' : 'Editor Restored');
      });
    }
  })();
