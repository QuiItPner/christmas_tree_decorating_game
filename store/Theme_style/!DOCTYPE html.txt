<script>
    // State management for Theme preview and mixer
    const mixerState = {
      foliageColor: '#0E472A',
      needleType: 'Douglas Fir',
      snowPercent: 45
    };

    function showToast(message) {
      const toast = document.getElementById('toastNotification');
      const toastMsg = document.getElementById('toastMessage');
      if (!toast || !toastMsg) return;

      toastMsg.innerText = message;
      toast.classList.remove('translate-y-24', 'opacity-0');
      toast.classList.add('translate-y-0', 'opacity-100');

      setTimeout(() => {
        toast.classList.remove('translate-y-0', 'opacity-100');
        toast.classList.add('translate-y-24', 'opacity-0');
      }, 3000);
    }

    function applyTheme(themeName) {
      showToast(`'${themeName}' applied to your Christmas Tree!`);
      
      // Update mixer visualizer to reflect chosen preset
      if (themeName === 'Classic Heritage') {
        setTreeFoliage('#005131', 'Douglas Fir');
        updateSnow(20);
      } else if (themeName === 'Winter Wonderland') {
        setTreeFoliage('#A4F4C3', 'Nordmann Blue');
        updateSnow(95);
      } else if (themeName === 'Pastel Sugar Plum') {
        setTreeFoliage('#FDFBF7', 'Blizzard Powder');
        updateSnow(40);
      } else if (themeName === 'Nordic Forest & Nature') {
        setTreeFoliage('#0E472A', 'Golden Cedar');
        updateSnow(15);
      } else if (themeName === 'Gingerbread Bakery') {
        setTreeFoliage('#5E4100', 'Cinnamon Twig');
        updateSnow(35);
      } else if (themeName === 'Midnight Starlight') {
        setTreeFoliage('#0E472A', 'Nordmann Blue');
        updateSnow(10);
      }
    }

    function customizeTheme(themeName) {
      showToast(`Opening '${themeName}' in Tree Studio customizer...`);
      const mixerSection = document.getElementById('snowSlider');
      if (mixerSection) {
        mixerSection.scrollIntoView({ behavior: 'smooth' });
      }
    }

    function setTreeFoliage(colorHex, name) {
      mixerState.foliageColor = colorHex;
      const tTop = document.getElementById('foliageTop');
      const tMid = document.getElementById('foliageMid');
      const tBot = document.getElementById('foliageBottom');
      const hexLabel = document.getElementById('selectedFolioHex');
      const summary = document.getElementById('summaryPresetLabel');

      if (tTop) tTop.setAttribute('fill', colorHex);
      if (tMid) tMid.setAttribute('fill', colorHex);
      if (tBot) tBot.setAttribute('fill', colorHex);
      if (hexLabel) hexLabel.innerText = colorHex;
      if (summary) summary.innerText = `${mixerState.needleType} (${colorHex})`;
    }

    function selectNeedle(button, needleType) {
      mixerState.needleType = needleType;
      document.querySelectorAll('.needle-btn').forEach(btn => {
        btn.classList.remove('bg-primary', 'text-on-primary');
        btn.classList.add('bg-surface-cream', 'text-on-surface');
      });
      button.classList.remove('bg-surface-cream', 'text-on-surface');
      button.classList.add('bg-primary', 'text-on-primary');

      const tag = document.getElementById('previewTag');
      const summary = document.getElementById('summaryPresetLabel');
      if (tag) tag.innerText = needleType;
      if (summary) summary.innerText = `${needleType} (${mixerState.foliageColor})`;
    }

    function updateSnow(val) {
      mixerState.snowPercent = val;
      const display = document.getElementById('snowLevelDisplay');
      const snowGroup = document.getElementById('snowCaps');
      const slider = document.getElementById('snowSlider');

      if (slider && slider.value !== val) slider.value = val;

      let descriptor = 'Light Powder';
      if (val > 70) descriptor = 'Blizzard Whiteout';
      else if (val > 40) descriptor = 'Gentle Flurry';
      else if (val == 0) descriptor = 'Snow Free';

      if (display) display.innerText = `${val}% (${descriptor})`;
      if (snowGroup) {
        snowGroup.style.opacity = (val / 100).toString();
      }
    }

    function resetMixer() {
      setTreeFoliage('#0E472A', 'Alpine Deep');
      updateSnow(45);
      const firstNeedle = document.querySelector('.needle-btn');
      if (firstNeedle) selectNeedle(firstNeedle, 'Douglas Fir');
      showToast('Swatches reset to holiday default!');
    }

    function launchCustomizedStudio() {
      showToast('Loading custom foliage configuration into Tree Studio...');
      setTimeout(() => {
        const studioNav = document.querySelector('nav a[data-path="tree-studio"]');
        if (studioNav) {
          studioNav.click();
        }
      }, 700);
    }
  </script>
